const SHEET_NAME = "Ratings"; // Serve the HTML app when opened directly function doGet(e) { // If called with ?action=get it's a data request (JSONP) if (e.parameter && e.parameter.action === "get") { return handleGet(e); } // Otherwise serve the web app return HtmlService .createHtmlOutputFromFile("index") .setTitle("AniRate") .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function handleGet(e) { try { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const ratings = []; if (data.length > 1) { const headers = data[0]; data.slice(1).forEach(row => { const obj = {}; headers.forEach((h, i) => { obj[h] = row[i]; }); ["plotRating","charactersRating","worldBuildingRating", "animationRating","soundtrackRating","enjoymentRating", "pacingRating","emotionalImpactRating","overallScore"].forEach(k => { obj[k] = parseFloat(obj[k]) || 0; }); obj.animeId = parseInt(obj.animeId) || 0; ratings.push(obj); }); } const payload = JSON.stringify({ ratings }); const cb = e.parameter && e.parameter.callback; if (cb) { return ContentService .createTextOutput(cb + "(" + payload + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(payload) .setMimeType(ContentService.MimeType.JSON); } catch(err) { const p = JSON.stringify({ error: err.message }); const cb = e.parameter && e.parameter.callback; if (cb) return ContentService.createTextOutput(cb+"("+p+")").setMimeType(ContentService.MimeType.JAVASCRIPT); return ContentService.createTextOutput(p).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.action !== "save") return respond({ error: "Unknown action" }); const sheet = getSheet(); const animeId = parseInt(body.animeId); const data = sheet.getDataRange().getValues(); let existingRow = -1; for (let i = 1; i < data.length; i++) { if (parseInt(data[i][0]) === animeId) { existingRow = i + 1; break; } } const row = [ animeId, body.animeTitle, body.plotRating, body.charactersRating, body.worldBuildingRating, body.animationRating, body.soundtrackRating, body.enjoymentRating, body.pacingRating, body.emotionalImpactRating, body.overallScore, body.timestamp || new Date().toISOString(), ]; if (existingRow > 0) { sheet.getRange(existingRow, 1, 1, row.length).setValues([row]); } else { sheet.appendRow(row); } return respond({ success: true }); } catch(err) { return respond({ error: err.message }); } } function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow([ "animeId","animeTitle","plotRating","charactersRating","worldBuildingRating", "animationRating","soundtrackRating","enjoymentRating", "pacingRating","emotionalImpactRating","overallScore","timestamp" ]); sheet.getRange(1,1,1,12).setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } function respond(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
